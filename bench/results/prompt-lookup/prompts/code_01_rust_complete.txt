use std::collections::HashMap;

/// A least-recently-used cache with a fixed capacity.
pub struct LruCache<K, V> {
    cap: usize,
    map: HashMap<K, (V, u64)>,
    tick: u64,
}

impl<K: std::hash::Hash + Eq + Clone, V> LruCache<K, V> {
    pub fn new(cap: usize) -> Self {
        Self { cap, map: HashMap::new(), tick: 0 }
    }

        self.tick += 1;

            Some(e) => {
                e.1 = t;
                Some(&e.0)
            }
            None => None,
        }
